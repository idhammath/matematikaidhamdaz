Nilai \displaystyle\lim_{x\to\frac{\pi}2}\dfrac{\sin\left(x-\dfrac{\pi}2\right)}{\sqrt{\dfrac{x}2}-\sqrt{\dfrac{\pi}4}} adalah
  1. 4\sqrt{\pi}
  2. 2\sqrt{\pi}
  3. \sqrt{\pi}
  1. \dfrac12\sqrt{\pi}
  2. \dfrac14\sqrt{\pi}
SUMBER
ALTERNATIF PENYELESAIAN
\begin{aligned} \displaystyle\lim_{x\to\frac{\pi}2}\dfrac{\sin\left(x-\dfrac{\pi}2\right)}{\sqrt{\dfrac{x}2}-\sqrt{\dfrac{\pi}4}}\color{red}{\cdot\dfrac{\sqrt{\dfrac{x}2}+\sqrt{\dfrac{\pi}4}}{\sqrt{\dfrac{x}2}+\sqrt{\dfrac{\pi}4}}}&=\displaystyle\lim_{x\to\frac{\pi}2}\dfrac{\left(\sqrt{\dfrac{x}2}+\sqrt{\dfrac{\pi}4}\right)\sin\left(x-\dfrac{\pi}2\right)}{\dfrac{x}2-\dfrac{\pi}4}\\[3.7pt] &=\displaystyle\lim_{x\to\frac{\pi}2}\dfrac{\left(\sqrt{\dfrac{x}2}+\sqrt{\dfrac{\pi}4}\right)\sin\left(x-\dfrac{\pi}2\right)}{\dfrac12\left(x-\dfrac{\pi}2\right)}\\[3.7pt] &=\displaystyle\lim_{x\to\frac{\pi}2}2\left(\sqrt{\dfrac{x}2}+\sqrt{\dfrac{\pi}4}\right)\cdot\dfrac{\sin\left(x-\dfrac{\pi}2\right)}{x-\dfrac{\pi}2}\\[3.7pt] &=2\left(\sqrt{\dfrac{\dfrac{\pi}2}2}+\sqrt{\dfrac{\pi}4}\right)\cdot1\\[3.7pt] &=2\left(\sqrt{\dfrac{\pi}4}+\sqrt{\dfrac{\pi}4}\right)\\[3.7pt] &=2\left(\dfrac12\sqrt{\pi}+\dfrac12\sqrt{\pi}\right)\\ &=\boxed{\boxed{2\sqrt{\pi}}} \end{aligned}
Jadi, {\displaystyle\lim_{x\to\frac{\pi}2}\dfrac{\sin\left(x-\dfrac{\pi}2\right)}{\sqrt{\dfrac{x}2}-\sqrt{\dfrac{\pi}4}}=2\sqrt{\pi}}.

  2. 6

ALTERNATIF PENYELESAIAN
\begin{aligned} \displaystyle\lim_{x\to0}\dfrac{3x}{\left(1-\sqrt{\cos2x}\right)\csc2x}&=\displaystyle\lim_{x\to0}\dfrac{3x\sin2x}{1-\sqrt{\cos2x}}\cdot{\color{red}\dfrac{1+\sqrt{\cos2x}}{1+\sqrt{\cos2x}}}\\[3.7pt] &=\displaystyle\lim_{x\to0}\dfrac{3x\sin2x\left(1+\sqrt{\cos2x}\right)}{1-\cos2x}\\[3.7pt] &=\displaystyle\lim_{x\to0}\dfrac{3x\sin2x\left(1+\sqrt{\cos2x}\right)}{1-\left(1-2\sin^2x\right)}\\[3.7pt] &=\displaystyle\lim_{x\to0}\dfrac{3x\sin2x\left(1+\sqrt{\cos2x}\right)}{1-1+2\sin^2x}\\[3.7pt] &=\displaystyle\lim_{x\to0}\dfrac{3x\sin2x\left(1+\sqrt{\cos2x}\right)}{2\sin^2x}\\[3.7pt] &=\displaystyle\lim_{x\to0}\dfrac32\cdot\dfrac{x}{\sin x}\cdot\dfrac{\sin2x}{\sin x}\cdot\left(1+\sqrt{\cos2x}\right)\\[3.7pt] &=\dfrac32\cdot\dfrac11\cdot\dfrac21\cdot\left(1+\sqrt{\cos2(0)}\right)\\[3.7pt] &=3\left(1+\sqrt{\cos0}\right)\\ &=3\left(1+\sqrt1\right)\\ &=3(1+1)\\ &=3(2)\\ &=\boxed{\boxed{6}} \end{aligned}
Jadi, \displaystyle\lim_{x\to0}\dfrac{3x}{\left(1-\sqrt{\cos2x}\right)\csc2x}=6.

ALTERNATIF PENYELESAIAN
\begin{aligned} \displaystyle\lim_{x\to0}\sqrt{\dfrac{x\cdot\tan x}{1+\sin^2x-\cos2x}}&=\displaystyle\lim_{x\to0}\sqrt{\dfrac{x\cdot\tan x}{1+\sin^2x-\left(1-2\sin^2x\right)}}\\[3.7pt] &=\displaystyle\lim_{x\to0}\sqrt{\dfrac{x\cdot\tan x}{1+\sin^2x-1+2\sin^2x}}\\[3.7pt] &=\displaystyle\lim_{x\to0}\sqrt{\dfrac{x\cdot\tan x}{3\sin^2x}}\\[3.7pt] &=\displaystyle\lim_{x\to0}\sqrt{\dfrac13\cdot\dfrac{x}{\sin x}\cdot\dfrac{\tan x}{\sin x}}\\[3.7pt] &=\sqrt{\dfrac13}\\[3.7pt] &=\dfrac1{\sqrt3}\\ &=\boxed{\boxed{\dfrac13\sqrt3}} \end{aligned}
Jadi, \displaystyle\lim_{x\to0}\sqrt{\dfrac{x\cdot\tan x}{1+\sin^2x-\cos2x}}=\dfrac13\sqrt3.

SIMAK UI 2011
ALTERNATIF PENYELESAIAN
\begin{aligned} \displaystyle\lim_{a\to b}\dfrac{\tan a-\tan b}{1+\left(1-\dfrac{a}b\right)\tan a\tan b-\dfrac{a}b}&=\displaystyle\lim_{a\to b}\dfrac{\tan a-\tan b}{1-\dfrac{a}b+\left(1-\dfrac{a}b\right)\tan a\tan b}\\ &=\displaystyle\lim_{a\to b}\dfrac{\tan a-\tan b}{\left(1-\dfrac{a}b\right)\left(1+\tan a\tan b\right)}\\ &=\displaystyle\lim_{a\to b}\dfrac{\tan( a-b)}{\dfrac{b-a}b}\\ &=\displaystyle\lim_{a\to b}\dfrac{\tan( a-b)}{\dfrac{a-b}{-b}}\\ &=\displaystyle\lim_{a\to b}\dfrac{-b\tan( a-b)}{a-b} \end{aligned} Misal a − b = x
Jika a ⟶ b maka x = a − b ⟶ b − b = 0 \begin{aligned} \displaystyle\lim_{a\to b}\dfrac{-b\tan( a-b)}{a-b}&=\displaystyle\lim_{x\to0}\dfrac{-b\tan x}x\\ &=\displaystyle\lim_{x\to0}-b\cdot\dfrac{tan x}x\\ &=-b\cdot1\\ &=\boxed{\boxed{-b}} \end{aligned}
Jadi, \displaystyle\lim_{a\to b}\dfrac{\tan a-\tan b}{1+\left(1-\dfrac{a}b\right)\tan a\tan b-\dfrac{a}b}=-b.
